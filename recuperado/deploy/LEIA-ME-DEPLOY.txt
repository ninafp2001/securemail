#   fly status -a security-webmail

#   Ver logs (erros):
#   fly logs -a security-webmail
#
#   Trocar senha do painel:
#   fly secrets set PANEL_PASS=OutraSenha123 -a security-webmail
#
#   Trocar usuario do painel:
#   fly secrets set PANEL_USER=OutroNome -a security-webmail
#
#   Reiniciar maquina:
#   fly apps restart security-webmail
#
#   Entrar na maquina (SSH):
#   fly ssh console -a security-webmail
#
#   Baixar arquivos da maquina para o PC:
#   fly ssh sftp get -a security-webmail -R /app/index.php C:\temp\index.php
